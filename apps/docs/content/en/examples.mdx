## Nested Fields

<Callout>Use dot notation (e.g., `"user.name"`) to work with nested structures</Callout>

Example of fields with a nested structure:

```tsx
import { z } from "zod";
import { field, Form, useForm, zormyResolver } from "zormy";

const UserNameField = field("user.name")
	.schema(z.string().min(3, "Name must be at least 3 characters"))
	.render(({ register, fieldState }) => (
		<div>
			<label>User Name</label>
			<input {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

const UserEmailField = field("user.email")
	.schema(z.string().email("Invalid email"))
	.render(({ register, fieldState }) => (
		<div>
			<label>User Email</label>
			<input type="email" {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

function NestedFieldsForm() {
	const form = useForm({
		resolver: zormyResolver({ fields: [UserNameField, UserEmailField] }),
		defaultValues: { user: { name: "", email: "" } },
	});

	return (
		<Form
			methods={form}
			onSubmit={form.handleSubmit((data) => alert(JSON.stringify(data, null, 2)))}
		>
			<UserNameField />
			<UserEmailField />
			<button type="submit">Submit</button>
		</Form>
	);
}
```

## Multi-Step Wizard

<Callout type="info">
	Wizards allow you to split complex forms into smaller, more manageable steps. Use
	`createWizardConfig` and `createWizardComponents` for strong typing.
</Callout>

Complete example of a multi-step wizard:

```tsx
import { z } from "zod";
import { createWizardComponents, createWizardConfig, field, useWizard } from "zormy";

const NameField = field("name")
	.schema(z.string().min(3, "Name must be at least 3 characters"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Name</label>
			<input {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

const EmailField = field("email")
	.schema(z.string().email("Invalid email"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Email</label>
			<input type="email" {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

const PasswordField = field("password")
	.schema(z.string().min(8, "Password must be at least 8 characters"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Password</label>
			<input type="password" {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

// Configure the wizard (outside the component to keep references stable)
const wizardConfig = createWizardConfig({
	steps: ["personal", "credentials"] as const,
	fields: {
		personal: [NameField],
		credentials: [EmailField, PasswordField],
	},
});

const { Wizard, Step } = createWizardComponents(wizardConfig);

function MultiStepWizard() {
	const wizard = useWizard({
		...wizardConfig,
		defaultValues: {
			name: "",
			email: "",
			password: "",
		},
		onComplete: (data) => {
			alert(JSON.stringify(data, null, 2));
		},
	});

	return (
		<Wizard methods={wizard} onSubmit={wizard.handleSubmit()}>
			<div>
				<p>
					Step {wizard.currentStepIndex + 1} of {wizard.totalSteps}
				</p>
				<p>Current step: {wizard.currentStep}</p>
			</div>

			<Step step="personal">
				<NameField />
			</Step>

			<Step step="credentials">
				<EmailField />
				<PasswordField />
			</Step>

			<div style={{ marginTop: "1rem" }}>
				{!wizard.isFirstStep && (
					<button type="button" onClick={wizard.back} style={{ marginRight: "0.5rem" }}>
						Back
					</button>
				)}
				{wizard.isLastStep ? (
					<button type="submit">Finish</button>
				) : (
					<button type="button" onClick={wizard.next}>
						Next
					</button>
				)}
			</div>
		</Wizard>
	);
}
```

## Conditional Step Wizard

Example of a wizard with steps that appear/disappear based on values:

```tsx
import { z } from "zod";
import { createWizardComponents, createWizardConfig, field, useWizard } from "zormy";

const WantToContributeField = field("wantToContribute")
	.schema(z.boolean())
	.render(({ register }) => (
		<div>
			<label>
				<input type="checkbox" {...register()} />
				Want to contribute?
			</label>
		</div>
	));

const RefuseReasonField = field("refuseReason")
	.schema(z.string().min(1, "Required field"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Refusal reason</label>
			<textarea {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

const SelectQuotaField = field("selectQuota")
	.schema(z.string().min(1, "Required field"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Select quota</label>
			<select {...register()}>
				<option value="">Select...</option>
				<option value="quota1">Quota 1</option>
				<option value="quota2">Quota 2</option>
			</select>
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

const config = createWizardConfig({
	steps: ["wantToContribute", "refuseReason", "selectQuota"] as const,
	fields: {
		wantToContribute: [WantToContributeField],
		refuseReason: [RefuseReasonField],
		selectQuota: [SelectQuotaField],
	},
	shouldIncludeStep: (step, formValues) => {
		if (step === "wantToContribute") return true;
		if (step === "refuseReason") return !formValues.wantToContribute;
		if (step === "selectQuota") return formValues.wantToContribute === true;
		return true;
	},
});

const { Wizard, Step } = createWizardComponents(config);

function ConditionalWizard() {
	const wizard = useWizard({
		...config,
		defaultValues: {
			wantToContribute: false,
			refuseReason: "",
			selectQuota: "",
		},
		onComplete: (data) => {
			alert(JSON.stringify(data, null, 2));
		},
	});

	return (
		<Wizard methods={wizard} onSubmit={wizard.handleSubmit()}>
			<div>
				<p>
					Step {wizard.currentStepIndex + 1} of {wizard.totalSteps}
				</p>
			</div>

			<Step step="wantToContribute">
				<WantToContributeField />
			</Step>

			<Step step="refuseReason">
				<RefuseReasonField />
			</Step>

			<Step step="selectQuota">
				<SelectQuotaField />
			</Step>

			<div style={{ marginTop: "1rem" }}>
				{!wizard.isFirstStep && (
					<button type="button" onClick={wizard.back} style={{ marginRight: "0.5rem" }}>
						Back
					</button>
				)}
				{wizard.isLastStep ? (
					<button type="submit">Finish</button>
				) : (
					<button type="button" onClick={wizard.next}>
						Next
					</button>
				)}
			</div>
		</Wizard>
	);
}
```

## Auto Save

Example of a wizard with auto-save:

```tsx
import { z } from "zod";
import {
	createWizardComponents,
	createWizardConfig,
	field,
	useAutoSaveContext,
	useWizard,
} from "zormy";

const NameField = field("name")
	.schema(z.string().min(3, "Name must be at least 3 characters"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Name</label>
			<input {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

const EmailField = field("email")
	.schema(z.string().email("Invalid email"))
	.render(({ register, fieldState }) => (
		<div>
			<label>Email</label>
			<input type="email" {...register()} />
			{fieldState.error && <span style={{ color: "red" }}>{fieldState.error.message}</span>}
		</div>
	));

function AutoSaveIndicator() {
	const autoSave = useAutoSaveContext();

	return (
		<div style={{ marginBottom: "1rem", fontSize: "0.875rem" }}>
			{autoSave.status === "saving" && <span>Saving...</span>}
			{autoSave.status === "saved" && <span style={{ color: "green" }}>Saved</span>}
			{autoSave.status === "error" && <span style={{ color: "red" }}>Error saving</span>}
			{autoSave.lastSaved && (
				<span style={{ marginLeft: "0.5rem", color: "#666" }}>
					(Saved at {autoSave.lastSaved.toLocaleTimeString()})
				</span>
			)}
		</div>
	);
}

const wizardConfig = createWizardConfig({
	steps: ["personal", "contact"] as const,
	fields: {
		personal: [NameField],
		contact: [EmailField],
	},
});

const { Wizard, Step } = createWizardComponents(wizardConfig);

function AutoSaveWizard() {
	const wizard = useWizard({
		...wizardConfig,
		defaultValues: {
			name: "",
			email: "",
		},
		autoSave: async ({ step, validChangedData }) => {
			// Simulates API call
			await new Promise((resolve) => setTimeout(resolve, 1000));
			console.log(`Saving step ${step}:`, validChangedData);
		},
		onComplete: (data) => {
			alert(JSON.stringify(data, null, 2));
		},
	});

	return (
		<Wizard methods={wizard} onSubmit={wizard.handleSubmit()}>
			<AutoSaveIndicator />

			<div>
				<p>
					Step {wizard.currentStepIndex + 1} of {wizard.totalSteps}
				</p>
			</div>

			<Step step="personal">
				<NameField />
			</Step>

			<Step step="contact">
				<EmailField />
			</Step>

			<div style={{ marginTop: "1rem" }}>
				{!wizard.isFirstStep && (
					<button type="button" onClick={wizard.back} style={{ marginRight: "0.5rem" }}>
						Back
					</button>
				)}
				{wizard.isLastStep ? (
					<button type="submit">Finish</button>
				) : (
					<button type="button" onClick={wizard.next}>
						Next
					</button>
				)}
			</div>
		</Wizard>
	);
}
```
